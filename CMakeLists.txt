cmake_minimum_required(VERSION 3.20)

if (NOT DEFINED ARCH)
    set(ARCH "x86_64" CACHE STRING "Kernel target architecture")
endif()

set_property(CACHE ARCH PROPERTY STRINGS x86_64 aarch64)

if (ARCH STREQUAL "x86_64")
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/toolchains/x86_64.cmake")
    project(FeatherOS LANGUAGES C CXX ASM_NASM)
    set(CMAKE_ASM_NASM_OBJECT_FORMAT elf64)
    add_compile_definitions(ARCH_x86_64)
elseif (ARCH STREQUAL "aarch64")
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/toolchains/aarch64.cmake")
    project(FeatherOS LANGUAGES C CXX ASM)
    add_compile_definitions(ARCH_aarch64)
else ()
    message(FATAL_ERROR "Unsupported ARCH='${ARCH}'. Choose x86_64 or aarch64.")
endif ()

set(KERNEL_BINARY "kernel.elf")
set(CMAKE_EXPORT_COMPILE_COMMANDS 1)
option(FEATHEROS_PANIC_TEST "Trigger an intentional kernel panic for smoke testing" OFF)
option(FEATHEROS_EXCEPTION_TEST "Trigger an intentional CPU exception for smoke testing" OFF)
option(FEATHEROS_INTERRUPT_TEST "Trigger a hardware interrupt for smoke testing" OFF)
option(FEATHEROS_MEMORY_TEST "Run physical and virtual memory tests" OFF)
option(FEATHEROS_TIMER_TEST "Run periodic timer tests" OFF)
option(FEATHEROS_SCHEDULER_TEST "Run cooperative scheduler tests" OFF)
option(FEATHEROS_USER_PREEMPTION_TEST "Run user-mode timer preemption tests" OFF)
option(FEATHEROS_SMP_TEST "Run multi-CPU scheduling tests" OFF)
option(FEATHEROS_STORAGE_TEST "Run block-device write and persistence tests" OFF)
option(FEATHEROS_NETWORK_TEST "Run network device and ARP tests" OFF)

add_executable(${KERNEL_BINARY})
if (FEATHEROS_PANIC_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_PANIC_TEST)
endif ()
if (FEATHEROS_EXCEPTION_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_EXCEPTION_TEST)
endif ()
if (FEATHEROS_INTERRUPT_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_INTERRUPT_TEST)
endif ()
if (FEATHEROS_MEMORY_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_MEMORY_TEST)
endif ()
if (FEATHEROS_TIMER_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_TIMER_TEST)
endif ()
if (FEATHEROS_SCHEDULER_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_SCHEDULER_TEST)
endif ()
if (FEATHEROS_USER_PREEMPTION_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_USER_PREEMPTION_TEST)
endif ()
if (FEATHEROS_SMP_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_SMP_TEST)
endif ()
if (FEATHEROS_STORAGE_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_STORAGE_TEST)
endif ()
if (FEATHEROS_NETWORK_TEST)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_NETWORK_TEST)
endif ()
if (FEATHEROS_PANIC_TEST OR FEATHEROS_EXCEPTION_TEST OR FEATHEROS_INTERRUPT_TEST OR FEATHEROS_MEMORY_TEST OR FEATHEROS_TIMER_TEST OR FEATHEROS_SCHEDULER_TEST OR FEATHEROS_USER_PREEMPTION_TEST OR FEATHEROS_SMP_TEST OR FEATHEROS_STORAGE_TEST OR FEATHEROS_NETWORK_TEST)
    set(FEATHEROS_KERNEL_TEST ON)
    target_compile_definitions(${KERNEL_BINARY} PRIVATE FEATHEROS_KERNEL_TEST)
endif ()

set(SRCROOT ${PROJECT_SOURCE_DIR})

target_include_directories(${KERNEL_BINARY} PRIVATE ${SRCROOT} ${SRCROOT}/deps/stivale ${SRCROOT}/src)

add_subdirectory(${SRCROOT}/deps)
add_subdirectory(${SRCROOT}/src)

target_link_libraries(${KERNEL_BINARY} PRIVATE limine)
